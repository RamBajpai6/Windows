variables:
  SSH_PRIVATE_KEY: "$SSH_PRIVATE_KEY"

before_script:
  - apt-get update -y && apt-get install -y openssh-client
  - mkdir -p ~/.ssh
  - echo "$SSH_PRIVATE_KEY" > ~/.ssh/id_rsa
  - chmod 400 ~/.ssh/id_rsa

deploy_to_server:
  stage: deploy
  script:
    - scp -i ~/.ssh/id_rsa -o StrictHostKeyChecking=no index.html ec2-user@<EC2-PublicIP>:/var/www/html/
  only:
    - main

before_script:
  - apt-get update -y && apt-get install -y openssh-client
  - mkdir -p ~/.ssh
  - echo "$SSH_PRIVATE_KEY" | tr -d '\r' > ~/.ssh/id_rsa
  - chmod 400 ~/.ssh/id_rsa
  - ssh-keyscan -H 13.234.202.227 >> ~/.ssh/known_hosts

deploy_to_server:
  stage: deploy
  script:
    - scp -i ~/.ssh/id_rsa index.html ec2-user@13.234.202.227:/var/www/html/
  only:
    - main
